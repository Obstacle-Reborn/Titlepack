#RequireContext CSmAction

#Include "MathLib" as ML

// =====================================================================
// Trampoline action — encoding v3. FROZEN once built into the Gbx.
// The mode arms us via ActionSetVariant (0 = disarmed). Read Variant only
// through its low 16 bits (see "Armed" in main): the high half of the word
// holds EnergyMax, so the raw value is never 0 and never in range.
//   Armed = Base + Up*1000 + Down*100 + Dir*10 + Horiz
//   Up/Down/Horiz = blast strength 0..9 (0 = no bullet on that component)
//   Base 10000 (C_VariantBase)    -> Dir 0..7 is a WORLD direction, straight
//                                    out of C_CompassDir (index i = i*45 deg
//                                    from +Z toward +X). Trampoline pads.
//   Base 20000 (C_VariantBaseRel) -> Dir 0..7 is RELATIVE to where the owner
//                                    looks: their aim yaw turned by i*45 deg,
//                                    in the same rotation sense. Dir 0 is
//                                    straight ahead. The mode's key-triggered
//                                    launch abilities.
// Rotation sense, N<->world calibration and the both-vertical policy live
// MODE-SIDE (Constants::C_TrampolineCompass*, C_LaunchRelativeMirror,
// C_TrampolineAllowBothVertical), so this Gbx never needs a rebuild for
// calibration or policy changes.
// C_VariantBase / C_VariantBaseRel must match
// Constants::C_TrampolineVariantBase / C_TrampolineVariantBaseRelative.
// State_Boolean1 = already fired (reset when the mode zeroes the variant).
// State_Integer1 = gauge initialized (see main): an instance the mode binds
// to a key must be activatable, or the press never reaches the mode.
// State_Integer2 = IsActive on the previous tick, for the press echo.
// =====================================================================
#Const C_VariantBase    10000
#Const C_VariantMax     19989	// base + 9*1000 + 9*100 + 9*10 + 9
#Const C_VariantBaseRel 20000
#Const C_VariantMaxRel  29989

// Bound instances are gated by their energy gauge and cooldown, so give them a
// free self-reloading one (pattern proven by Actions/Sneak.Script.txt). Firing
// itself stays variant-driven and unaffected — this only decides whether pressing
// the key registers at all. The small cooldown just thins out input events; the
// launch cooldown the mapper sets is enforced mode-side.
#Const C_BoundCooldown	100
#Const C_BoundEnergy		6400

// Projectile models, index = strength - 1. All: self-exploding (very short
// lifetime), Blow Radius Attenuation 0 (position-independent force).
// TrampV1-9: high Blow Vertical Scale. TrampH1-9: Blow Vertical Scale 0,
// higher overall strengths.
#Const C_VerticalProjectiles [
	"TrampV1", "TrampV2", "TrampV3", "TrampV4", "TrampV5",
	"TrampV6", "TrampV7", "TrampV8", "TrampV9"
]
#Const C_HorizontalProjectiles [
	"TrampH1", "TrampH2", "TrampH3", "TrampH4", "TrampH5",
	"TrampH6", "TrampH7", "TrampH8", "TrampH9"
]
#Const C_FallbackProjectile "TrampV1"

// One shared "effect" projectile fired once per launch (any variant) to carry the
// launch feedback: particle burst + sound. Author it once in the Action Maker as a
// harmless bullet — Blow strength 0, Damage 0, Explode-On-All-Players OFF, very short
// lifetime — so it never pushes or hurts anyone and never stacks with the directional
// bullets. NOT routed through GetProjectileId (whose fallback is a real launch bullet):
// until this model exists the effect is simply skipped, never double-launching.
#Const C_EffectProjectile "TrampFX"

// Compass table: index i = i*45 deg, rotating from world +Z toward +X.
// Which index the editor's "N" means is the mode's business, not ours.
// Relative launches (C_VariantBaseRel) use the same steps around the owner's
// aim instead of around +Z, computed continuously below.
#Const C_CompassDir [
	<0., 0., 1.>,  <0.7071, 0., 0.7071>,   <1., 0., 0.>,  <0.7071, 0., -0.7071>,
	<0., 0., -1.>, <-0.7071, 0., -0.7071>, <-1., 0., 0.>, <-0.7071, 0., 0.7071>
]

// Tuning knobs (world units)
#Const C_HorizontalOffset	0.8		// how far beside the player the horizontal bullet spawns
#Const C_HorizontalHeight	0.5		// height above the feet for the horizontal bullet
#Const C_UpwardOffset		0.15	// just above the feet — anything below ground level detonates
									// under the floor surface and the blast does nothing
#Const C_DownwardOffset		2.2		// above the head, shooting up, for downward launches
#Const C_EffectHeight		1.0		// height above the feet for the TrampFX feedback bullet

Ident GetProjectileId(Text _ModelName) {
	declare ModelId = Projectile_GetModelId(_ModelName);
	if (ModelId == NullId) {
		ModelId = Projectile_GetModelId(C_FallbackProjectile);
	}
	return ModelId;
}

main() {
	// First tick: make this instance usable as a bound key (harmless when unbound)
	if (State_Integer1 != 1) {
		State_Integer1 = 1;
		Cooldown = C_BoundCooldown;
		EnergyMax = C_BoundEnergy;
		EnergyCost = 0;
		EnergyReload = True;
		Energy = EnergyMax;
	}

	// A bound instance is a key-triggered launch ability: report the key press to the mode
	// ourselves. The engine's mode-side OnActionEvent is not delivered for custom actions —
	// Nadeo's shield action and Actions/Sneak.Script.txt report activation the same way. The
	// mode maps the echo back to a key through the event's ShooterActionSlot. Pad instances
	// are never bound, so they never send this.
	// State_Integer2 = IsActive on the previous tick.
	if (IsBound) {
		declare WasActive = State_Integer2 == 1;
		if (IsActive && !WasActive) {
			State_Integer2 = 1;
			SendRulesEvent("trampoline", ["press"], Owner, Null);
		} else if (!IsActive && WasActive) {
			State_Integer2 = 0;
		}
	}

	// What the mode armed us with. ActionSetVariant writes only the low 16 bits of the word
	// Variant reads back — its high half holds EnergyMax — so an instance with a gauge reads
	// EnergyMax*65536 + the value the mode wrote. Without masking that off, this action reads
	// e.g. 419445400 for a launch of 15000: never 0, so the fired-latch below never resets, and
	// never in range, so it echoes without ever creating a projectile.
	declare Armed = Variant % 65536;

	if (Armed == 0) {
		State_Boolean1 = False;
	} else if (!State_Boolean1 && Owner != Null && Owner.SpawnStatus == CSmPlayer::ESpawnStatus::Spawned) {
		State_Boolean1 = True;

		// Echo the fired variant to the mode: it disarms us (resets the variant) and closes the PvP window
		SendRulesEvent("trampoline", [""^Armed], Owner, Null);

		declare Relative = Armed >= C_VariantBaseRel && Armed <= C_VariantMaxRel;
		if ((Armed >= C_VariantBase && Armed <= C_VariantMax) || Relative) {
			declare Base = C_VariantBase;
			if (Relative) {
				Base = C_VariantBaseRel;
			}
			declare Packed  = Armed - Base;
			declare Up      = Packed / 1000;
			declare Down    = (Packed / 100) % 10;
			declare Dir     = (Packed / 10) % 10;
			declare Horiz   = Packed % 10;
			if (Dir > 7) {
				Dir = 0;	// defensive: only 0..7 is ever sent
			}

			declare LastProjectileId = NullId;

			// Horizontal push: self-exploding bullet beside the player,
			// opposite the launch direction, aimed/moving outward.
			if (Horiz > 0) {
				declare LaunchDir = C_CompassDir[Dir];
				if (Relative) {
					// Continuous version of the compass table, turned by the picked step:
					// the horizontal yaw the owner is looking at, plus Dir*45 deg. Sin/Cos
					// of the yaw avoids normalizing (the flattened aim shrinks as the player
					// looks up or down); looking exactly straight up or down leaves Atan2 at
					// 0, so the push falls back to the step measured from +Z instead of
					// collapsing to nothing.
					declare Yaw = ML::Atan2(Owner.AimDirection.X, Owner.AimDirection.Z) + Dir * (ML::PI() / 4.);
					LaunchDir = <ML::Sin(Yaw), 0., ML::Cos(Yaw)>;
				}
				declare Out = <-LaunchDir.X, 0., -LaunchDir.Z>;
				declare Position = Owner.Position + <Out.X * C_HorizontalOffset, C_HorizontalHeight, Out.Z * C_HorizontalOffset>;
				LastProjectileId = Projectile_CreateAtLocation(GetProjectileId(C_HorizontalProjectiles[Horiz - 1]), Null, Position, Out, Out);
			}

			// Vertical pushes: both may fire; the MODE decides the policy
			// (it zeroes the down digit when both-vertical is disallowed).
			if (Up > 0) {
				declare Position = Owner.Position + <0., C_UpwardOffset, 0.>;
				LastProjectileId = Projectile_CreateAtLocation(GetProjectileId(C_VerticalProjectiles[Up - 1]), Null, Position, <0., -1., 0.>, <0., -1., 0.>);
			}
			if (Down > 0) {
				declare Position = Owner.Position + <0., C_DownwardOffset, 0.>;
				LastProjectileId = Projectile_CreateAtLocation(GetProjectileId(C_VerticalProjectiles[Down - 1]), Null, Position, <0., 1., 0.>, <0., 1., 0.>);
			}

			// Launch feedback (particles + sound), fired last so the directional bullets
			// above are created first — though all spawn on the same tick (main() never
			// yields), so this doesn't delay them. One shot per launch, so it can't stack
			// the way per-bullet effects do. Skipped while the model is absent — no
			// GetProjectileId fallback, so no accidental extra launch. Explodes in place
			// (zero velocity); direction is irrelevant but kept non-zero.
			declare EffectModelId = Projectile_GetModelId(C_EffectProjectile);
			if (EffectModelId != NullId) {
				declare EffectPosition = Owner.Position + <0., C_EffectHeight, 0.>;
				// capture the returned Ident (like the directional calls) so it isn't discarded
				LastProjectileId = Projectile_CreateAtLocation(EffectModelId, Null, EffectPosition, <0., 1., 0.>, <0., 0., 0.>);
			}

			State_EntityId1 = LastProjectileId;
		}
	}
}
