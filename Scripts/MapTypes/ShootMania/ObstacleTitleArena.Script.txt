// metadata names can't be composed at runtime, so all four blocks are spelled out once in

// The eight launch directions, laid out clockwise from "straight ahead" — they are relative to
// where the player looks, not compass directions, so the picker shows arrows (up = forward) and
// the mode turns the player's aim by Dir*45 deg.
// Mid line of the direction picker bar: the clip window and every arrow are centred on it, so a
// rotated arrow stays on the same spot instead of drifting up or down.
#Const C_LaunchDirBarMid "-2.55"

Boolean IsEnabled(CAnchorData _Landmark)
{
    declare metadata Meta_Enabled for _Landmark = True;
    return Meta_Enabled;
}


Boolean IsGoal(CAnchorData _Landmark)
{
    if (!IsEnabled(_Landmark))
        return False;

    declare metadata Meta_CPType for _Landmark = 0;
    return _Landmark.Tag == "Goal" || Meta_CPType == 2;
}


Boolean IsCheckpoint(CAnchorData _Landmark)
{
    if (!IsEnabled(_Landmark))
        return False;

    declare metadata Meta_CPType for _Landmark = 0;
    return _Landmark.Tag == "Checkpoint" && (Meta_CPType == 0 || Meta_CPType == 1);
}


Boolean IsPortal(CAnchorData _Landmark)
{
    if (!IsEnabled(_Landmark))
        return False;

    declare metadata Meta_IsPortal for _Landmark = False;
    return Meta_IsPortal;
}


Boolean IsCpSpawn(CAnchorData _Landmark)
{
    if (!IsEnabled(_Landmark))
        return False;

    declare metadata Meta_IsCpSpawn for _Landmark = False;
    return Meta_IsCpSpawn;
}

// Same bar as Select8, but the options are arrows instead of words: one arrow icon turned by
// 45° per slot (clockwise from "straight ahead"). The icon quad is used rather than an arrow
// character because the editor font has no diagonal arrows (they render as empty rhombuses) and
// its glyphs are not centred in their box, which throws a rotated arrow off its slot.
// Ids are _IdPrefix ^ index (0..7), so the click handler reads the picked index off the id.
Text Select8Arrows(Text _Position, Text _IdPrefix, Text _Selector)
{
    declare Options = "";
    for (Index, 0, 7) {
        declare ArrowId = _IdPrefix ^ Index;
        declare ArrowX = ""^(-35 + Index * 10);
        // The icon points right, so slot 0 ("straight ahead") starts a quarter turn back.
        declare ArrowRot = ""^(Index * 45 - 90);
        // The turn is applied to a wrapper frame, not to the quad: a rotated control spins around
        // its position anchor, while a frame spins around its own origin - which the centred quad
        // sits exactly on, so every arrow turns in place. The quad stays small enough that its
        // rotated corners keep clear of the clip window.
        Options ^= """
            <frame pos="{{{ArrowX}}} {{{C_LaunchDirBarMid}}}" rot="{{{ArrowRot}}}" z-index="1">
                <quad id="{{{ArrowId}}}" pos="0 0" size="4 4" halign="center" valign="center"
                    keepratio="Fit" style="UIConstruction_Buttons" substyle="Right" scriptevents="1"/>
            </frame>
        """;
    }

    return """
        <frame pos="{{{_Position}}}">
            <frame clip="True" clipposn="0 {{{C_LaunchDirBarMid}}}" clipsizen="80 6" halign="center">
                {{{Options}}}
                <quad  id="{{{_Selector}}}" pos="-40 1"   size="10 7" bgcolor="4777C1FF"/>
            </frame>
            <quad pos="0 0.5" size="81 6.1" bgcolor="797979AA" halign="center"/>
        </frame>
    """;
}

Void Init()
{
    if (MapType::GetVersion() == 0) { // new map
        declare metadata Integer Meta_ForceFastRS for Map = 1;
        MapType::SetVersion(3);
    }
    CustomEditAnchorData = True;

    foreach (Anchor in AnchorData)
        NewAnchorCache.add(Anchor.Id);
    G_AnchorCache = NewAnchorCache;
}

Void CopyLaunchMeta(CAnchorData _From, CAnchorData _To)
{
    for (Key, 1, Constants::C_LaunchKeyCount) {
        LaunchMetaWrite(_To, Key, LaunchMetaRead(_From, Key));
    }

// The four launch-ability keys on the checkpoint/spawn dialogs, laid out like the weapon
// picker: one shared bar with the key numbers as the options, and every granted key sitting on
// the same blue box the weapon picker uses to mark its selection (several may be lit at once).
// Baked at open time: the only way to change a key is the sub-window, which reopens this dialog.
Text LaunchKeyPicker(CAnchorData _Anchor, Text _Position)
{
    // Half the width of a Select* bar — single digits need far less room than weapon names.
    declare Total = 40;
    declare Count = Constants::C_LaunchKeyCount;
    declare Width = Total / Count;
    declare TextTotal = ""^Total;
    declare TextBar = ""^(Total + 1);

    declare Labels = "";
    declare Highlights = "";
    for (Key, 1, Count) {
        declare CenterX = -Total / 2 + Width / 2 + (Key - 1) * Width;
        declare TextKey = ""^Key;
        declare TextCenterX = ""^CenterX;
        declare TextLeftX = ""^(CenterX - Width / 2);
        declare TextWidth = ""^Width;

        declare Values = LaunchMetaRead(_Anchor, Key);
        if (Values[C_LaunchFieldOn] == 1) {
            Highlights ^= """<quad pos="{{{TextLeftX}}} 1" size="{{{TextWidth}}} 7" bgcolor="4777C1FF"/>""";
        }
        Labels ^= """
            <label id="Launch_Key{{{TextKey}}}" pos="{{{TextCenterX}}} -2" size="{{{TextWidth}}} 6"
                text="{{{TextKey}}}" halign="center" valign="center" textsize="2" class="text" scriptevents="1"/>
        """;
    }

    declare Picker = """
        <frame pos="0 -3.5">
            <frame clip="True" clipposn="0 -2.5" clipsizen="{{{TextTotal}}} 5" halign="center">
                {{{Labels}}}
                {{{Highlights}}}
            </frame>
            <quad pos="0 0.5" size="{{{TextBar}}} 6.1" bgcolor="797979AA" halign="center"/>
        </frame>
    """;

    return BigProperty(_Position, "Action Keys", Picker);
}

                            }
                        }
                    }
                    case CMlScriptEvent::Type::EntrySubmit: {
                        declare Entry <=> (Event.Control as CMlEntry);
                        declare EntryValue = ParseTime(Entry.Value);
                        
                        switch(Entry.ControlId) {
                            case "Entry_Champion_Medal": Champion_Medal = EntryValue;
                            case "Entry_Gold_Medal": Gold_Medal = EntryValue;
                            case "Entry_Silver_Medal": Silver_Medal = EntryValue;
                            case "Entry_Bronze_Medal": Bronze_Medal = EntryValue;
                        }

                        if (Gold_Medal < Champion_Medal || Gold_Medal <= 0) {
                            Entry_Gold_Medal.TextColor = CL::HexToRgb("e40");
                        } else {
                            Entry_Gold_Medal.TextColor = CL::HexToRgb("fff");
                        }
                        if (Silver_Medal < Gold_Medal || Silver_Medal <= 0) {
                            Entry_Silver_Medal.TextColor = CL::HexToRgb("e40");
                        } else {
                            Entry_Silver_Medal.TextColor = CL::HexToRgb("fff");
                        }
                        if (Bronze_Medal < Silver_Medal || Bronze_Medal <= 0) {
                            Entry_Bronze_Medal.TextColor = CL::HexToRgb("e40");
                        } else {
                            Entry_Bronze_Medal.TextColor = CL::HexToRgb("fff");
                        }

                        NeedUpdate = True;
                    }
                }
            }
        }
    }
    --></script>
    </manialink>
    """;
}

Void SaveObjectives()
{
    // We don't use to the Map#TMObjective_{X}Time because only the author time correspond
    // to the time we provide. Other times are calculated based on the author time, so they
    // don't correspond to what we provide. So we use meta variables for them. These
    // meta variables are those used accross the whole mode to determine the medal times
    // defined by the author of a map.

    // FIXME(ahmad): Actually, we can't trust only the medal times meta variables.
    // On old versions of the Obstacle map type, the simple fact of opening the "Edit objectives"
    // window, could edit these meta variables, and e.g. set the author time to 1s (which is
    // the default value of the author time objective). So we rely on this meta variable.
    // This is bad because old maps that defined medal times are seen as not having medal times :(
    declare metadata Boolean Meta_IsMedalSet for Map = False;

    declare metadata Integer Meta_Champion_Medal for Map = -1;
    declare metadata Integer Meta_Gold_Medal for Map = -1;
    declare metadata Integer Meta_Silver_Medal for Map = -1;
    declare metadata Integer Meta_Bronze_Medal for Map = -1;

    declare Integer Champion_Medal for ManialinkPage = -1;
    declare Integer Gold_Medal for ManialinkPage = -1;
    declare Integer Silver_Medal for ManialinkPage = -1;
    declare Integer Bronze_Medal for ManialinkPage = -1;
    
    if (Champion_Medal > 0) {
        Meta_Champion_Medal = Champion_Medal;
        Meta_IsMedalSet = True;
    }
    if (Gold_Medal > 0 && Gold_Medal > Champion_Medal) {
        Meta_Gold_Medal = Gold_Medal;
        Meta_IsMedalSet = True;
    }
    if (Silver_Medal > 0 && Silver_Medal > Gold_Medal) {
        Meta_Silver_Medal = Silver_Medal;
        Meta_IsMedalSet = True;
    }
    if (Bronze_Medal > 0 && Bronze_Medal > Silver_Medal) {
        Meta_Bronze_Medal = Bronze_Medal;
        Meta_IsMedalSet = True;
    }
}

Void ClearObjectives() {
    declare metadata Boolean Meta_IsMedalSet for Map = False;
    declare metadata Integer Meta_Champion_Medal for Map = -1;
    declare metadata Integer Meta_Gold_Medal for Map = -1;
    declare metadata Integer Meta_Silver_Medal for Map = -1;
    declare metadata Integer Meta_Bronze_Medal for Map = -1;

    Meta_Champion_Medal = -1;
    Meta_Gold_Medal = -1;
    Meta_Silver_Medal = -1;
    Meta_Bronze_Medal = -1;
    Meta_IsMedalSet = False;
}

Void PreserveObjectives()
{
    if (Map.TMObjective_AuthorTime == -1) {
        declare metadata Integer Meta_Champion_Medal for Map = 1000;
        declare metadata Integer Meta_Gold_Medal for Map = 1000;
        declare metadata Integer Meta_Silver_Medal for Map = 1000;
        declare metadata Integer Meta_Bronze_Medal for Map = 1000;
        
        Map.TMObjective_AuthorTime = Meta_Champion_Medal;
        Map.TMObjective_GoldTime = Meta_Gold_Medal;
        Map.TMObjective_SilverTime = Meta_Silver_Medal;
        Map.TMObjective_BronzeTime = Meta_Bronze_Medal;
    }
}

Void EditLaunchAbility(CAnchorData _Anchor, Integer _Key)
{
    /// --- Variables
    G_LaunchEditAnchor = _Anchor.Id;
    G_LaunchEditKey = _Key;

    declare Boolean LaunchOn for ManialinkPage = False;
    declare Integer LaunchUp for ManialinkPage = 0;
    declare Integer LaunchDown for ManialinkPage = 0;
    declare Integer LaunchSide for ManialinkPage = 0;
    declare Integer LaunchDir for ManialinkPage = 0;
    declare Integer LaunchCooldown for ManialinkPage = 0;
    declare Boolean LaunchGround for ManialinkPage = False;

    declare Values = LaunchMetaRead(_Anchor, _Key);
    LaunchOn = Values[C_LaunchFieldOn] == 1;
    LaunchUp = Values[C_LaunchFieldUp];
    LaunchDown = Values[C_LaunchFieldDown];
    LaunchSide = Values[C_LaunchFieldSide];
    LaunchDir = Values[C_LaunchFieldDir];
    LaunchCooldown = Values[C_LaunchFieldCooldown];
    LaunchGround = Values[C_LaunchFieldGround] == 1;

    /// --- Manialink
    declare Properties = """
        <frame id="Basic" pos="0 25">
            {{{Select2("0 -2.5", "Enable", "Disable", "LaunchOn_Selector")}}}
            {{{BigProperty("0 -13", "Strength", "")}}}
            {{{InlineProperty("10 -19.5", "Upward", PlusMinus("7 1.25", "+", "UpPlus") ^ PlusMinus("7 -1.25", "-", "UpMinus") ^ Entry("-1 0", "Entry_LaunchUp"))}}}
            {{{InlineProperty("10 -27.5", "Downward", PlusMinus("7 1.25", "+", "DownPlus") ^ PlusMinus("7 -1.25", "-", "DownMinus") ^ Entry("-1 0", "Entry_LaunchDown"))}}}
            {{{InlineProperty("10 -35.5", "Horizontal", PlusMinus("7 1.25", "+", "SidePlus") ^ PlusMinus("7 -1.25", "-", "SideMinus") ^ Entry("-1 0", "Entry_LaunchSide"))}}}
            {{{BigProperty("0 -46", "Direction", Select8Arrows("0 -4", C_LaunchDirIdPrefix, "Launch_Dir_Selector"))}}}
            {{{BigProperty("0 -63", "Cooldown", Entry("0 -6", "Entry_LaunchCooldown"))}}}
            <label pos="13 -69" size="10 4" text="ms" textsize="1" class="text" halign="left" valign="center2"/>
            <label pos="-3 -77" size="52 5" text="Only when touching ground" textsize="2" class="text" halign="center" valign="center2"/>
            {{{Checkbox("21 -77", "Checkbox_LaunchGround")}}}
        </frame>
    """;

    SetWindow(MakeWindowWithBack("Action Key "^_Key, Properties));

    ManialinkText = """
    <manialink version="3" name="Obstacle:EditLaunchAbility">
    {{{GetML()}}}
    <script><!--
    #Include "TextLib" as TL

    Integer Clamp0_9(Integer _Value) {
        if (_Value < 0) {
            return 0;
        }
        if (_Value > 9) {
            return 9;
        }
        return _Value;
    }

    Integer ClampCooldown(Integer _Value) {
        if (_Value < 0) {
            return 0;
        }
        if (_Value > {{{TL::ToText(Constants::C_LaunchCooldownMax)}}}) {
            return {{{TL::ToText(Constants::C_LaunchCooldownMax)}}};
        }
        return _Value;
    }

    Void UpdateView() {
        declare Boolean LaunchOn for Page;
        declare Integer LaunchUp for Page;
        declare Integer LaunchDown for Page;
        declare Integer LaunchSide for Page;
        declare Integer LaunchDir for Page;
        declare Integer LaunchCooldown for Page;
        declare Boolean LaunchGround for Page;

        declare Entry_LaunchUp <=> (Page.GetFirstChild("Entry_LaunchUp") as CMlEntry);
        Entry_LaunchUp.SetText(TL::ToText(LaunchUp), False);
        declare Entry_LaunchDown <=> (Page.GetFirstChild("Entry_LaunchDown") as CMlEntry);
        Entry_LaunchDown.SetText(TL::ToText(LaunchDown), False);
        declare Entry_LaunchSide <=> (Page.GetFirstChild("Entry_LaunchSide") as CMlEntry);
        Entry_LaunchSide.SetText(TL::ToText(LaunchSide), False);
        declare Entry_LaunchCooldown <=> (Page.GetFirstChild("Entry_LaunchCooldown") as CMlEntry);
        Entry_LaunchCooldown.SetText(TL::ToText(LaunchCooldown), False);

        // Same Enable / Disable pair the checkpoint window uses for its own Enabled flag
        declare Quad_LaunchOn_Selector <=> (Page.GetFirstChild("LaunchOn_Selector") as CMlQuad);
        if (LaunchOn) {
            Quad_LaunchOn_Selector.RelativePosition_V3.X = -20.;
        } else {
            Quad_LaunchOn_Selector.RelativePosition_V3.X = 0.;
        }

        declare Label_LaunchGround <=> (Page.GetFirstChild("Checkbox_LaunchGround") as CMlLabel);
        if (LaunchGround && Label_LaunchGround.Value != "✖") {
            Label_LaunchGround.SetText("✖");
        }
        if (!LaunchGround && Label_LaunchGround.Value != "") {
            Label_LaunchGround.SetText("");
        }

        declare Quad_Dir_Selector <=> (Page.GetFirstChild("Launch_Dir_Selector") as CMlQuad);
        Quad_Dir_Selector.RelativePosition_V3.X = -40. + LaunchDir*10.;

        SendCustomEvent("Event_Save_LaunchAbility", Text[]);
    }

    main() {
        declare Boolean LaunchOn for Page;
        declare Integer LaunchUp for Page;
        declare Integer LaunchDown for Page;
        declare Integer LaunchSide for Page;
        declare Integer LaunchDir for Page;
        declare Integer LaunchCooldown for Page;
        declare Boolean LaunchGround for Page;
        declare NeedUpdate = False;
        UpdateView();

        while (True) {
            yield;

            if (NeedUpdate) {
                UpdateView();
                NeedUpdate = False;
            }

            foreach (Event in PendingEvents) {
                switch (Event.Type){
                    case CMlScriptEvent::Type::MouseClick: {
                        // Going back must not trigger one more save from this page: the plugin
                        // flushes the page itself, and by then it no longer points at this key.
                        declare Leaving = False;
                        switch (Event.ControlId) {
                            case "Window_Background":
                                SendCustomEvent("Event_Property_Window_Clicked", Text[]);
                            case "Launch_Back": {
                                SendCustomEvent("Event_Launch_Back", Text[]);
                                Leaving = True;
                            }
                            case "Enable":
                                LaunchOn = True;
                            case "Disable":
                                LaunchOn = False;
                            case "Checkbox_LaunchGround":
                                LaunchGround = !LaunchGround;
                            case "UpPlus":
                                LaunchUp = Clamp0_9(LaunchUp + 1);
                            case "UpMinus":
                                LaunchUp = Clamp0_9(LaunchUp - 1);
                            case "DownPlus":
                                LaunchDown = Clamp0_9(LaunchDown + 1);
                            case "DownMinus":
                                LaunchDown = Clamp0_9(LaunchDown - 1);
                            case "SidePlus":
                                LaunchSide = Clamp0_9(LaunchSide + 1);
                            case "SideMinus":
                                LaunchSide = Clamp0_9(LaunchSide - 1);
                            // Direction arrows, clockwise from "straight ahead" (see Select8Arrows)
                            case "{{{C_LaunchDirIdPrefix}}}0":
                                LaunchDir = 0;
                            case "{{{C_LaunchDirIdPrefix}}}1":
                                LaunchDir = 1;
                            case "{{{C_LaunchDirIdPrefix}}}2":
                                LaunchDir = 2;
                            case "{{{C_LaunchDirIdPrefix}}}3":
                                LaunchDir = 3;
                            case "{{{C_LaunchDirIdPrefix}}}4":
                                LaunchDir = 4;
                            case "{{{C_LaunchDirIdPrefix}}}5":
                                LaunchDir = 5;
                            case "{{{C_LaunchDirIdPrefix}}}6":
                                LaunchDir = 6;
                            case "{{{C_LaunchDirIdPrefix}}}7":
                                LaunchDir = 7;
                        }

                        if (!Leaving) {
                            NeedUpdate = True;
                        }
                    }
                    case CMlScriptEvent::Type::EntrySubmit: {
                        declare Entry <=> (Event.Control as CMlEntry);
                        switch(Entry.ControlId) {
                            case "Entry_LaunchUp":
                                LaunchUp = Clamp0_9(TL::ToInteger(Entry.Value));
                            case "Entry_LaunchDown":
                                LaunchDown = Clamp0_9(TL::ToInteger(Entry.Value));
                            case "Entry_LaunchSide":
                                LaunchSide = Clamp0_9(TL::ToInteger(Entry.Value));
                            case "Entry_LaunchCooldown":
                                LaunchCooldown = ClampCooldown(TL::ToInteger(Entry.Value));
                        }
                        NeedUpdate = True;
                    }
                }
            }
        }
    }
    --></script>
    </manialink>
    """;
}


Void SaveLaunchAbilitySettings()
{
    // Write to the anchor the open sub-window was opened for, not to whatever anchor the
    // editor currently considers selected: the page belongs to G_LaunchEditAnchor/Key by
    // construction, so it can never spill its values onto another anchor, and the save keeps
    // working if something clears the selection while the window is up.
    if (G_LaunchEditKey <= 0) {
        return;
    }
    if (!AnchorData.existskey(G_LaunchEditAnchor)) {
        return;
    }
    declare Anchor <=> AnchorData[G_LaunchEditAnchor];

    declare Boolean LaunchOn for ManialinkPage = False;
    declare Integer LaunchUp for ManialinkPage = 0;
    declare Integer LaunchDown for ManialinkPage = 0;
    declare Integer LaunchSide for ManialinkPage = 0;
    declare Integer LaunchDir for ManialinkPage = 0;
    declare Integer LaunchCooldown for ManialinkPage = 0;
    declare Boolean LaunchGround for ManialinkPage = False;

    LaunchMetaWrite(Anchor, G_LaunchEditKey, [
        LaunchFlag(LaunchOn), LaunchUp, LaunchDown, LaunchSide,
        LaunchDir, LaunchCooldown, LaunchFlag(LaunchGround)
    ]);
}

{

        Quad_Compass_Selector.RelativePosition_V3.X = -40. + TrampCompass*10.;

        SendCustomEvent("Event_Save_TrampolineItem", Text[]);
    }

    main() {
        declare Integer TrampUp for Page;
        declare Integer TrampDown for Page;
        declare Integer TrampHoriz for Page;
        declare Integer TrampCompass for Page;
        declare Boolean TrampWaitGround for Page;
        declare Integer TrampDelay for Page;
        declare Integer TrampCooldown for Page;
        // Prefill bookkeeping: TrampDelayPrefilled stays True only while the delay still holds
        // the value the "wait until touching ground" tick wrote, so unticking can put back
        // TrampDelayBeforePrefill without ever discarding a delay the mapper typed themselves.
        declare Boolean TrampDelayPrefilled for Page;
        declare Integer TrampDelayBeforePrefill for Page;
        declare NeedUpdate = False;
        UpdateView();

        while (True) {

            if (NeedUpdate) {
                UpdateView();
                NeedUpdate = False;
            }

            foreach (Event in PendingEvents) {

                                TrampUp = Clamp0_9(TrampUp - 1);
                                NeedUpdate = True;
                            }
                            case "DownPlus": {
                                TrampDown = Clamp0_9(TrampDown + 1);
                                NeedUpdate = True;
                            }
                            case "DownMinus": {
                                TrampDown = Clamp0_9(TrampDown - 1);
                                NeedUpdate = True;
                            }
                            case "HorizPlus": {
                                TrampHoriz = Clamp0_9(TrampHoriz + 1);
                                NeedUpdate = True;
                            }
                            case "HorizMinus": {
                                TrampHoriz = Clamp0_9(TrampHoriz - 1);
                                NeedUpdate = True;
                            }
                            case "N": {
                                TrampCompass = 0;
                                NeedUpdate = True;
                            }
                            case "NE": {
                                TrampCompass = 1;
                                NeedUpdate = True;
                            }
                            case "E": {
                                TrampCompass = 2;
                                NeedUpdate = True;
                            }
                            case "SE": {
                                TrampCompass = 3;
                                NeedUpdate = True;
                            }
                            case "S": {
                                TrampCompass = 4;
                                NeedUpdate = True;
                            }
                            case "SW": {
                                TrampCompass = 5;
                                NeedUpdate = True;
                            }
                            case "W": {
                                TrampCompass = 6;
                                NeedUpdate = True;
                            }
                            case "NW": {
                                TrampCompass = 7;
                                NeedUpdate = True;
                            }
                            case "Checkbox_TrampWaitGround": {
                                TrampWaitGround = !TrampWaitGround;
                                if (TrampWaitGround) {
                                    // A wait-ground pad needs a small delay to fire after the
                                    // landing, so prefill one -- but never shorten a delay that is
                                    // already at least as long as the prefill.
                                    if (TrampDelay < {{{TL::ToText(Constants::C_TrampolineWaitGroundDelayPrefill)}}}) {
                                        TrampDelayBeforePrefill = TrampDelay;
                                        TrampDelay = {{{TL::ToText(Constants::C_TrampolineWaitGroundDelayPrefill)}}};
                                        TrampDelayPrefilled = True;
                                    }
                                } else {
                                    // Untick undoes our own prefill only; a delay edited since the
                                    // tick has cleared the flag and is kept as-is.
                                    if (TrampDelayPrefilled) {
                                        TrampDelay = TrampDelayBeforePrefill;
                                        TrampDelayPrefilled = False;
                                    }
                                }
                                NeedUpdate = True;
                            }
                        }
                    }
                    case CMlScriptEvent::Type::EntrySubmit: {
                        declare Entry <=> (Event.Control as CMlEntry);
                        switch(Entry.ControlId) {
                            case "Entry_TrampUp": {
                                TrampUp = Clamp0_9(TL::ToInteger(Entry.Value));
                            }
                            case "Entry_TrampDown": {
                                TrampDown = Clamp0_9(TL::ToInteger(Entry.Value));
                            }
                            case "Entry_TrampHoriz": {
                                TrampHoriz = Clamp0_9(TL::ToInteger(Entry.Value));
                            }
                            case "Entry_TrampCooldown": {
                                TrampCooldown = ClampCooldown(TL::ToInteger(Entry.Value));
                            }
                            case "Entry_TrampDelay": {
                                TrampDelay = ClampDelay(TL::ToInteger(Entry.Value));
                                // The mapper owns the delay from here on: unticking must not
                                // revert it. Retyping the prefilled value itself is not an edit.
                                if (TrampDelay != {{{TL::ToText(Constants::C_TrampolineWaitGroundDelayPrefill)}}}) {
                                    TrampDelayPrefilled = False;
                                }
                            }
                        }
                        NeedUpdate = True;
                    }
                }
            }
        }
    }
    --></script>
    </manialink>
    """;
}

Void SaveTrampolineItemSettings()
{
    if (!AnchorData.existskey(G_LastAnchorId)) {
        return;
    }

    declare Anchor <=> AnchorData[G_LastAnchorId];
    // Accept every trampoline size family (DefaultTag = item model name containing "Trampoline").
    if (!TL::Find("Trampoline", Anchor.DefaultTag, False, True)) {
        return;
    }

    declare Integer TrampUp for ManialinkPage = 0;
    declare Integer TrampDown for ManialinkPage = 0;
    declare Integer TrampHoriz for ManialinkPage = 0;
    declare Integer TrampCompass for ManialinkPage = 0;
    declare Boolean TrampWaitGround for ManialinkPage = False;
    declare Integer TrampDelay for ManialinkPage = 0;
    declare Integer TrampCooldown for ManialinkPage = 0;

    declare metadata Meta_TrampolineUp for Anchor = Constants::C_TrampolineDefaultUp;
    declare metadata Meta_TrampolineDown for Anchor = Constants::C_TrampolineDefaultDown;
    declare metadata Meta_TrampolineCompass for Anchor = Constants::C_TrampolineDefaultCompass;
    declare metadata Meta_TrampolineHoriz for Anchor = Constants::C_TrampolineDefaultHoriz;
    declare metadata Meta_TrampolineWaitGround for Anchor = Constants::C_TrampolineDefaultWaitGround;

Void UpdateValidability()
{
    declare StartSpawns = 0;
    declare Goals = 0;
    declare Integer[Integer] CPCounts = [];

    foreach (Anchor in AnchorData)
    {
        if (!IsEnabled(Anchor))
            continue;

        switch (Anchor.Tag)
        {
            case "Spawn":
            {
                if (!IsCpSpawn(Anchor))
                {
                    StartSpawns += 1;
                    if (StartSpawns > 1)
                    {
                        ValidationStatus = CSmMapType::ValidationStatus::NotValidable;
                        ValidabilityRequirementsMessage = _("You must place exactly one Start Spawn.");
                        return;
                    }
                }
            }
            default :
            {
                if (IsGoal(Anchor))
                {
                    Goals += 1;
                }
                else if (IsCheckpoint(Anchor))
                {
                    if (!CPCounts.existskey(Anchor.Order)) {
                        CPCounts[Anchor.Order] = 0;
                    }
                    CPCounts[Anchor.Order] += 1;
                }
            }
        }
    }

    declare MaxOrder = 0;
    declare HasNegativeOrders = False;
    foreach (Order => Count in CPCounts) {
        MaxOrder = ML::Max(MaxOrder, Order);
        if (Order < 0) HasNegativeOrders = True;
    }


    if (StartSpawns == 0)
    {
        ValidationStatus = CSmMapType::ValidationStatus::NotValidable;
        ValidabilityRequirementsMessage = _("You must place at least one Start Spawn.");
    }
    else if (Goals == 0)
    {
        ValidationStatus = CSmMapType::ValidationStatus::NotValidable;
        ValidabilityRequirementsMessage = _("You must place at least one Goal.");
    }
    else if (CPCounts.existskey(0))
    {
        ValidationStatus = CSmMapType::ValidationStatus::NotValidable;
        ValidabilityRequirementsMessage = _("Every checkpoint needs to have an order.");
    }
    else
    {
        if (HasNegativeOrders) {
            ValidationStatus = CSmMapType::ValidationStatus::NotValidable;
            ValidabilityRequirementsMessage = "Checkpoints order have to be positive";
        } else {
            declare IsOk = True;
            declare Missing = -1;

            for (Order, 1, MaxOrder) {
                if (!CPCounts.existskey(Order)) {
                    Missing = Order;
                    IsOk = False;
                    break;
                }
            }

            if (IsOk) {
                ValidationStatus = CSmMapType::ValidationStatus::Validated;
            }
            else {
                ValidationStatus = CSmMapType::ValidationStatus::NotValidable;
                ValidabilityRequirementsMessage = "Missing checkpoint #"^Missing;
            }
        }
    }
}

Void TestWaypoint(Boolean IsCP)
{
    if (!AnchorData.existskey(G_LastAnchorId))
        return;
    

    declare Editor_WayPointSpawn_Flag for Mode = 0;
    declare Editor_WayPointSpawn_isCheckpoint for Mode = False;
    declare rng = ML::Rand(-2147483647, 2147483647);
    
    Editor_WayPointSpawn_Flag = rng;
    Editor_WayPointSpawn_isCheckpoint = IsCP;

    Editor_WayPointSpawn_isCheckpoint = False;
}


Void Loop()
{
    declare Ident CopiedSpawnId for ManialinkPage;
    declare Ident CopiedCheckpointId for ManialinkPage;

    foreach (Event in PendingEvents)
    {
        switch (Event.Type)
        {
            case CMapEditorPluginEvent::Type::MapModified:
            {

                if (G_AnchorCache.count != AnchorData.count)
                {
                    declare Ident[] NewAnchorCache;
                    declare FoundTheNewSpawn = False;
                    
                    declare CanPasteSpawn = False;
                    declare CanPasteCP = False;

                    if (CopiedSpawnId != NullId && AnchorData.existskey(CopiedSpawnId)) {
                        CanPasteSpawn = True;
                    } else {
                        CopiedSpawnId = NullId;
                    }

                    if (CopiedCheckpointId != NullId && AnchorData.existskey(CopiedCheckpointId)) {
                        CanPasteCP = True;
                    } else {
                        CopiedCheckpointId = NullId;
                    }

                    foreach (Anchor in AnchorData)
                    {
                        NewAnchorCache.add(Anchor.Id);

                        if (EditMode == CMapEditorPlugin::EditMode::Place && !FoundTheNewSpawn)
                        {
                            if (Anchor.Order == 0 && !G_AnchorCache.exists(Anchor.Id))
                            {
                                if ("Spawn" == Anchor.DefaultTag)
                                {
                                    if (CanPasteSpawn) {
                                        G_LastAnchorId = Anchor.Id;
                                        PasteSpawn(CopiedSpawnId);
                                        EditSpawn(Anchor);
                                    } else {
                                        declare AnySpawn = False;
                                        foreach (MaybeSpawn in AnchorData) {
                                            if (MaybeSpawn.DefaultTag == "Spawn" && MaybeSpawn != Anchor) {
                                                declare metadata Meta_IsCpSpawn for MaybeSpawn = False;
                                                if (!Meta_IsCpSpawn) {
                                                    AnySpawn = True;
                                                    break;
                                                }
                                            }
                                        }
                                        
                                        if (AnySpawn) {
                                            declare metadata Meta_IsCpSpawn for Anchor = False;
                                            Meta_IsCpSpawn = True;
                                        }
                                    }
                                } else if ("Checkpoint" == Anchor.DefaultTag && CanPasteCP) {
                                    G_LastAnchorId = Anchor.Id;
                                    PasteCheckpoint(CopiedCheckpointId);
                                    EditCheckpoint(Anchor);
                                }
                                
                                
                                FoundTheNewSpawn = True;
                            }
                        }
                    }

                    G_AnchorCache = NewAnchorCache;
                }
                UpdateValidability();
            }

            case CMapEditorPluginEvent::Type::EditAnchor:
            {
                if (!AnchorData.existskey(Event.EditedAnchorDataId))
                    continue;

                ManialinkText = "";
                CustomSelectionCoords.clear();

                declare Anchor <=> AnchorData[Event.EditedAnchorDataId];
                G_LastAnchorId = Anchor.Id;
								////////////////////
								if(TL::SubText(Anchor.Tag, 0, 2) == "Dy")	
								{
									HighlightAnchor(Anchor);
									EditDynamic(Anchor);
								}
								////////////////////

                // Trampoline launch-pads: DefaultTag equals the item's model name -- "Trampoline"
                // for the original pad and "<size> Trampoline[ Small]" for the Trampolines\\Normal|Small
                // families -- so every size is caught by the shared "Trampoline" substring.
                if (TL::Find("Trampoline", Anchor.DefaultTag, False, True)) {
                    EditTrampolineItem(Anchor);
                }

                switch (Anchor.DefaultTag)
                {
                    case "Goal": {
                        HighlightAnchor(Anchor);
                        EditGoal(Anchor);
                    }
                    case "Checkpoint": {
                        HighlightAnchor(Anchor);
                        EditCheckpoint(Anchor);
                    }
                    case "Spawn": {
                        HighlightAnchor(Anchor);
                        EditSpawn(Anchor);
                    }
                    case "Gate": {
                        //EditGate(Anchor);
                    }

                    case "Ammo":
                        EditAmmoItem(Anchor);
                    case "OffzoneShield":
                        EditOffzoneShieldItem(Anchor);
                    default :
                        continue;
                }
            }

            case CMapEditorPluginEvent::Type::StartValidation:
            {
                TestMapWithMode("Obstacle.Script.txt");
            }

            case CMapEditorPluginEvent::Type::EditObjectives:
            {
                EditObjectives();
            }

            case CMapEditorPluginEvent::Type::MapSavedOrSaveCancelled:
            {
                if (Event.MapSavedOrSaveCancelled) {
                    //
                }
            }

            case CMapEditorPluginEvent::Type::LayerCustomEvent:
            {
                switch (Event.CustomEventType)
                {
                    case "Close":
                        ManialinkText = "";
                    case "Event_Property_Window_Clicked":
                        if (PlaceMode != CMapEditorPlugin::PlaceMode::BlockProperty)
                            PlaceMode = CMapEditorPlugin::PlaceMode::BlockProperty;
                    case "Event_Edit": {
                        declare CurrentOrder = TL::ToInteger(Event.CustomEventData[1]);
                        declare Dir = TL::ToInteger(Event.CustomEventData[2]);

                        // Fill CPsOrder array based on AnchorData
                        declare CAnchorData[Integer] CPsOrder;
                        foreach (Anchor in AnchorData) {
                            if (Anchor.DefaultTag == Event.CustomEventData[0])
                                CPsOrder[Anchor.Order] <=> Anchor;
                        }
                        CPsOrder = CPsOrder.sortkey();

                        // Fill CPs array from CPsOrder and find the index of the current CP
                        declare CAnchorData[] CPs;
                        declare CurrentIndex = 0;
                        foreach (CP in CPsOrder) {
                            if (CP.Order == CurrentOrder) 
                                CurrentIndex = CPs.count;
                            CPs.add(CP);
                        }

                        // adding CPs.count prevents weird negative modulo stuff
                        declare Order = (CPs.count + CurrentIndex + Dir) % CPs.count;
                        declare NextAnchor <=> CPs[Order];

                        G_LastAnchorId = NextAnchor.Id;
                        CenterOnAnchor(NextAnchor);

                        if (Event.CustomEventData[0] == "Checkpoint") {
                            EditCheckpoint(NextAnchor);
                        } else {
                            EditSpawn(NextAnchor);
                        }
                    }

                    case "Event_Test_Checkpoint": {
                        TestWaypoint(True);
                    }
                    case "Event_Test_Spawn": {
                        TestWaypoint(False);
                    }

                    case "Event_Paste_Checkpoint": {
                        if (CopiedCheckpointId != NullId && CopiedCheckpointId != G_LastAnchorId) {
                            PasteCheckpoint(CopiedCheckpointId);
                        } 
                    }
                    case "Event_Paste_Spawn": {
                        if (CopiedSpawnId != NullId && CopiedSpawnId != G_LastAnchorId) {
                            PasteSpawn(CopiedSpawnId);
                        } 
                    }

                    case "Event_Copy_Checkpoint": {
                        CopiedCheckpointId = G_LastAnchorId;
                    }
                    case "Event_Copy_Spawn": {
                        CopiedSpawnId = G_LastAnchorId;
                    }

                    case "Event_Save": {
                        SaveMap(MapFileName);
                    }
                    case "Event_Save_Objectives": {
                        SaveObjectives();
                    }
                    case "Event_Clear_Objectives": {
                        ClearObjectives();
                    }
                    case "Event_Save_Map": {
                        SaveMapSettings();
                    }
                    case "Event_Save_Goal": {
                        SaveGoalSettings();
                    }
                    case "Event_Save_Checkpoint": {
                        SaveCheckpointSettings();
                    }
                    case "Event_Save_Spawn": {
                        SaveSpawnSettings();
                    }
                    case "Event_Save_AmmoItem": {
                        SaveAmmoItemSettings();
                    }
                    case "Event_Save_OffzoneShieldItem": {
                        SaveOffzoneShieldItemSettings();
                    }
                    case "Event_Save_TrampolineItem": {
                        SaveTrampolineItemSettings();
                    }
                    case "Event_Save_LaunchAbility": {
                        SaveLaunchAbilitySettings();
                    }
                    case "Event_Edit_Launch": {
                        // Flush whatever key is on screen before repointing the page at another
                        SaveLaunchAbilitySettings();
                        if (AnchorData.existskey(G_LastAnchorId)) {
                            declare LaunchAnchor <=> AnchorData[G_LastAnchorId];
                            EditLaunchAbility(LaunchAnchor, ML::Clamp(TL::ToInteger(Event.CustomEventData[0]), 1, Constants::C_LaunchKeyCount));
                        }
                    }
                    case "Event_Launch_Back": {
                        SaveLaunchAbilitySettings();
                        G_LaunchEditAnchor = NullId;
                        G_LaunchEditKey = 0;
                        if (AnchorData.existskey(G_LastAnchorId)) {
                            declare BackAnchor <=> AnchorData[G_LastAnchorId];
                            if (BackAnchor.DefaultTag == "Spawn") {
                                EditSpawn(BackAnchor);
                            } else {
                                EditCheckpoint(BackAnchor);
                            }
                        }
                    }
										case "Event_Save_ItemProperties": {
												SaveItemProperties();
										}
                }

                UpdateValidability();
            }

        }
    }
}


main()
{
    Load();
    Init();
    UpdateValidability();

    declare persistent P_EnableAirMapping for LocalUser = False;
    declare persistent P_EnableMixMapping for LocalUser = False;
    EnableAirMapping = P_EnableAirMapping;
    EnableMixMapping = P_EnableMixMapping;

    declare CMapEditorPlugin::PlaceMode PlaceModeCache;
    while (True)
    {
        PreserveObjectives();
        Loop();

        if (PlaceModeCache != PlaceMode) {
            PlaceModeCache = PlaceMode;

            if (PlaceMode == CMapEditorPlugin::PlaceMode::BlockProperty) {
                if (ManialinkText != "" && G_LastAnchorId != NullId) {
                    HighlightAnchor(AnchorData[G_LastAnchorId]);
                } else {
                    EditMap();
                }
            } else {
                CustomSelectionCoords.clear();
                ManialinkText = "";
            }
        }

        if (P_EnableAirMapping != EnableAirMapping || P_EnableMixMapping != EnableMixMapping) {
            P_EnableAirMapping = EnableAirMapping;
            P_EnableMixMapping = EnableMixMapping;
        }
        yield;
    }
    Unload();
}
